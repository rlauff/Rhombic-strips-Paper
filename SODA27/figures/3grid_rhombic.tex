
\begin{tikzpicture}
\def\hamcycle{0/0,1/0,2/0,3/0,3/1,3/2,2/2,2/1,1/1,1/2,0/2,0/1}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Left panel: Hamilton cycle F on C_{b1} x C_{b2}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{scope}
  \foreach \x in {0,...,2} \foreach \y in {0,...,2}
    \draw[gray!40] (1.2*\x,1.2*\y) -- (1.2*\x+1.2,1.2*\y);
  \foreach \x in {0,...,3} \foreach \y in {0,...,1}
    \draw[gray!40] (1.2*\x,1.2*\y) -- (1.2*\x,1.2*\y+1.2);
  \foreach \x/\y [remember=\x as \px (initially 0), remember=\y as \py (initially 1)]
      in \hamcycle
    \draw[Blue,very thick] (1.2*\px,1.2*\py) -- (1.2*\x,1.2*\y);
  \foreach \x in {0,...,3} \foreach \y in {0,...,2}
    \fill (1.2*\x,1.2*\y) circle (2pt);
  \node at (1.8,-0.9) {$C_{3}\times C_{2}$};
\end{scope}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Right panel: F' = (F x [0,b3]) + all z-edges
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{scope}[xshift=7.5cm]
  % oblique projection (x,y,z) -> (1.4x + 0.65y, 0.42y + 1.5z)
  \newcommand{\pt}[3]{({1.4*(#1)+0.65*(#2)},{0.42*(#2)+1.5*(#3)})}

  % faint grid: all x- and y-edges on each level
  \foreach \z in {0,...,2}{
    \foreach \x in {0,...,2} \foreach \y in {0,...,2}
      \draw[gray!35] \pt{\x}{\y}{\z} -- \pt{\x+1}{\y}{\z};
    \foreach \x in {0,...,3} \foreach \y in {0,...,1}
      \draw[gray!35] \pt{\x}{\y}{\z} -- \pt{\x}{\y+1}{\z};
  }

  % all vertical edges (in F')
  \foreach \x in {0,...,3} \foreach \y in {0,...,2} \foreach \z in {0,1}
    \draw[Purple,thick] \pt{\x}{\y}{\z} -- \pt{\x}{\y}{\z+1};

    \draw[Purple,thick] \pt{0}{1}{2} -- \pt{1}{1}{2};
    \draw[Purple,thick] \pt{1}{0}{2} -- \pt{1}{1}{2};
    \draw[Purple,thick] \pt{2}{0}{2} -- \pt{2}{1}{2};
    \draw[Purple,thick] \pt{2}{1}{2} -- \pt{3}{1}{2};
    \draw[Purple,thick] \pt{0}{1}{0} -- \pt{1}{1}{0};
    \draw[Purple,thick] \pt{1}{0}{0} -- \pt{1}{1}{0};
    \draw[Purple,thick] \pt{2}{0}{0} -- \pt{2}{1}{0};
    \draw[Purple,thick] \pt{2}{1}{0} -- \pt{3}{1}{0};

  % lifted copies of the cycle (in F')
  \foreach \z in {0,...,2}
    \foreach \x/\y [remember=\x as \px (initially 0), remember=\y as \py (initially 1)]
        in \hamcycle
      \draw[Blue,very thick] \pt{\px}{\py}{\z} -- \pt{\x}{\y}{\z};

  % vertices
  \foreach \z in {0,...,2} \foreach \x in {0,...,3} \foreach \y in {0,...,2}
    \fill \pt{\x}{\y}{\z} circle (1.8pt);

  \node at (3.0,-0.9) {$C_3\times C_2\times C_2$};
\end{scope}

\end{tikzpicture}
